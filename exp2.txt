#include<stdio.h>
#include<stdlib.h>
#include<string.h>

struct patient {
        int id ;
        char name[50];
        char status[50];
        struct patient *next;
};

struct patient *head = NULL;

void addpatient()
{
        struct patient *newnode = NULL;
        newnode = (struct patient*)malloc(sizeof(struct patient));

        printf("Enter patient ID : ");
        scanf("%d",&newnode -> id);

        getchar();

        printf("Enter patient Name : ");
        fgets(newnode->name, sizeof(newnode->name), stdin);

        newnode->name[strcspn(newnode->name, "\n")] = '\0';

        printf("Enter patient Status : ");
        fgets(newnode->status, sizeof(newnode->status), stdin);

        newnode->status[strcspn(newnode->status, "\n")] = '\0';

        newnode -> next = NULL;

        if(head == NULL)
        {
        head = newnode;
        }
        else
        {
        struct patient *temp = head;
        while(temp -> next != NULL)
        {
        temp = temp -> next;
        }

        temp -> next = newnode;

        }
        printf("Patient added successfully!!\n");

}

void displaypatients()
{
        if(head == NULL)
        {
        printf("No Patients Available");
        return;
        }

        printf("------ PATIENTS RECORD ------\n");

        struct patient *temp = head;

        while(temp != NULL)
        {
                printf("Patient id     : %d\n",temp -> id);
                printf("Patient Name   : %s\n",temp -> name);
                printf("Patient Status : %s\n",temp -> status);
                printf("--------------------------------------\n");
                temp = temp -> next;
        }

}


int main()
{
    int choice;

    while(1)
    {
        printf("\n===== HOSPITAL MANAGEMENT =====\n");
        printf("1. Add Patient\n");
        printf("2. Update Status\n");
        printf("3. Display Patients\n");
        printf("4. EXIT\n");

        printf("Enter your choice : ");
        scanf("%d",&choice);
        switch(choice)
        {
            case 1:
                addpatient();
                break;

            case 2:
                updatestatus();
                break;

            case 3:
                displaypatients();
                break;

            case 4:
