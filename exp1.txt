#include<stdio.h>
#define max 2147483647

int main()
{
        int num = 0;
        printf("Enter the number : ");
        scanf("%d",&num);
        int factorial = 1;

        while(num >= 1)
        {
                if(factorial > (max / num))
                {
                        printf("OVERFLOW !!\n");
                        return 0;
                }
                factorial *= num;
                num--;
        }
        printf("Factorial = %d\n",factorial);

        return 0;
}
